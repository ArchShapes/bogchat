""" Glitch Prank - a harmless fake "screen glitch" with distorted sound. How it works: it takes ONE screenshot of your screen (kept only in memory, never saved or sent anywhere), then shows that picture full-screen with tearing, shifted blocks, RGB splitting, inverted bands and scanlines, while playing synthesized glitchy sound. After 15 seconds it closes itself. It does NOT change any system settings, touch your files, lock your keyboard, or copy itself. Press Esc or Q at any time to stop. WARNINGS: * Fast flashing: don't run it near anyone sensitive to flashing lights. * Loud, harsh sound: turn your volume DOWN first and take off headphones. The script never changes your system volume - you control how loud it is. Setup (once): pip install pillow Run: python glitch_prank.py """ import math import os import random import struct import subprocess import sys import tempfile import time import tkinter as tk import wave try: from PIL import Image, ImageChops, ImageDraw, ImageGrab, ImageOps, ImageTk except ImportError: print("This needs Pillow. Install it with: pip install pillow") sys.exit(1) DURATION = 15 # seconds FRAME_MS = 70 # time between frames (raise for calmer visuals) SOUND_LEVEL = 0.7 # 0.0 - 1.0 (relative to your system volume) NEAREST = getattr(Image, "Resampling", Image).NEAREST # ---------------------------------------------------------------- sound ---- def make_sound(path, seconds=DURATION, rate=22050, level=SOUND_LEVEL): """Write a distorted, bit-crushed glitch sound to a WAV file.""" total = int(seconds * rate) samples = [] while len(samples) < total: seg = random.randint(int(rate * 0.03), int(rate * 0.25)) kind = random.choice(("square", "noise", "sweep", "saw", "silence")) freq = random.uniform(60, 1800) crush = random.choice((1, 4, 8, 16)) hold = 0.0 for n in range(seg): t = n / rate if kind == "square": v = 1.0 if math.sin(2 * math.pi * freq * t) > 0 else -1.0 elif kind == "noise": v = random.uniform(-1, 1) elif kind == "sweep": v = math.sin(2 * math.pi * freq * (1 + 4 * n / seg) * t) elif kind == "saw": v = ((freq * t) % 1.0) * 2 - 1 else: v = 0.0 v = max(-1.0, min(1.0, v * 2.5)) # hard clipping = distortion if n % crush == 0: # bit-crush / sample-hold hold = v samples.append(int(hold * level * 32767)) samples = samples[:total] with wave.open(path, "wb") as wf: wf.setnchannels(1) wf.setsampwidth(2) wf.setframerate(rate) wf.writeframes(struct.pack("<%dh" % len(samples), *samples)) class SoundPlayer: def __init__(self): self.path = os.path.join(tempfile.gettempdir(), "glitch_prank.wav") self.proc = None self.win = None try: import winsound self.win = winsound except ImportError: pass def start(self): try: make_sound(self.path) if self.win: self.win.PlaySound(self.path, self.win.SND_FILENAME | self.win.SND_ASYNC) elif sys.platform == "darwin": self.proc = subprocess.Popen(["afplay", self.path]) else: for cmd in ("paplay", "aplay"): try: self.proc = subprocess.Popen([cmd, self.path], stderr=subprocess.DEVNULL) break except FileNotFoundError: continue except Exception as e: print("Sound unavailable:", e) def stop(self): try: if self.win: self.win.PlaySound(None, self.win.SND_PURGE) if self.proc: self.proc.terminate() except Exception: pass try: os.remove(self.path) except Exception: pass # --------------------------------------------------------------- visuals --- def grab_screen(w, h): try: return ImageGrab.grab().convert("RGB").resize((w, h), NEAREST) except Exception: # Fallback if screenshots aren't allowed: random colored blocks img = Image.new("RGB", (w, h), (20, 20, 30)) d = ImageDraw.Draw(img) for _ in range(60): x, y = random.randint(0, w), random.randint(0, h) d.rectangle((x, y, x + random.randint(20, w // 3), y + random.randint(10, h // 5)), fill=tuple(random.randint(0, 255) for _ in range(3))) return img def glitch(base): w, h = base.size img = base.copy() # horizontal tearing: shift whole bands sideways (some inverted) for _ in range(random.randint(6, 18)): y = random.randint(0, h - 10) y2 = min(h, y + random.randint(4, max(5, h // 8))) band = img.crop((0, y, w, y2)) band = ImageChops.offset(band, random.randint(-w // 6, w // 6), 0) if random.random() < 0.2: band = ImageOps.invert(band) img.paste(band, (0, y)) # displaced blocks for _ in range(random.randint(3, 10)): bw = random.randint(w // 20, w // 4) bh = random.randint(h // 30, h // 6) x, y = random.randint(0, w - bw), random.randint(0, h - bh) block = img.crop((x, y, x + bw, y + bh)) nx = max(0, min(w - bw, x + random.randint(-w // 8, w // 8))) ny = max(0, min(h - bh, y + random.randint(-h // 10, h // 10))) img.paste(block, (nx, ny)) # RGB channel split if random.random() < 0.85: r, g, b = img.split() d = random.randint(3, 20) r = ImageChops.offset(r, d, 0) b = ImageChops.offset(b, -d, random.randint(-4, 4)) img = Image.merge("RGB", (r, g, b)) # occasional full inversion / color flash roll = random.random() if roll < 0.10: img = ImageOps.invert(img) elif roll < 0.18: tint = Image.new("RGB", img.size, tuple(random.randint(0, 255) for _ in range(3))) img = Image.blend(img, tint, 0.5) # scanlines d = ImageDraw.Draw(img) for y in range(0, h, 3): d.line((0, y, w, y), fill=(0, 0, 0)) return img # ------------------------------------------------------------------- app --- class GlitchPrank: def __init__(self): self.root = tk.Tk() sw, sh = self.root.winfo_screenwidth(), self.root.winfo_screenheight() self.root.withdraw() self.base = grab_screen(sw // 2, sh // 2) # work at half res for speed self.size = (sw, sh) self.root.title("Glitch") self.root.configure(bg="black") self.root.attributes("-fullscreen", True) self.root.attributes("-topmost", True) self.root.bind("", self.stop) self.root.bind("q", self.stop) self.root.bind("Q", self.stop) self.root.deiconify() self.label = tk.Label(self.root, bg="black", bd=0) self.label.pack(fill="both", expand=True) self.sound = SoundPlayer() self.sound.start() self.start = time.time() self.root.after(int((DURATION + 1) * 1000), self.stop) # safety net self.tick() def tick(self): if time.time() - self.start >= DURATION: self.stop() return frame = glitch(self.base).resize(self.size, NEAREST) self.photo = ImageTk.PhotoImage(frame) self.label.configure(image=self.photo) self.root.after(FRAME_MS, self.tick) def stop(self, event=None): self.sound.stop() try: self.root.destroy() except tk.TclError: pass def run(self): self.root.mainloop() if __name__ == "__main__": GlitchPrank().run()
